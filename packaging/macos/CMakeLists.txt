include(FetchContent)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.json" pins)
function(fetch_pinned name)
    string(JSON url GET "${pins}" ${name} url)
    string(JSON commit GET "${pins}" ${name} commit)
    # SOURCE_SUBDIR deliberately absent: populate only, then let upstream's
    # guarantee_* functions configure the SDK. No SDK examples/tests are built.
    FetchContent_Declare(${name} GIT_REPOSITORY "${url}" GIT_TAG "${commit}"
        GIT_SUBMODULES "" SOURCE_SUBDIR unused)
    FetchContent_MakeAvailable(${name})
    set(${name}_SOURCE_DIR "${${name}_SOURCE_DIR}" PARENT_SCOPE)
endfunction()
fetch_pinned(clap_sdk)
set(CLAP_SDK_ROOT "${clap_sdk_SOURCE_DIR}" CACHE PATH "Pinned CLAP" FORCE)
if(AU_FORMAT STREQUAL "auv2")
    fetch_pinned(audiounit_sdk)
    set(AUDIOUNIT_SDK_ROOT "${audiounit_sdk_SOURCE_DIR}" CACHE PATH "Pinned AUv2 SDK" FORCE)
elseif(NOT AU_FORMAT STREQUAL "auv3")
    message(FATAL_ERROR "AU_FORMAT must be auv2 or auv3")
